        // Pipeline Evaluator Engine
        function runPearlDetectionPipeline() {
            const npi = document.getElementById('providerNpi').value;
            const specialty = document.getElementById('specialty').value;
            const cpt = document.getElementById('cptCode').value;
            const billed = parseFloat(document.getElementById('billedAmount').value) || 0;
            const hours = parseFloat(document.getElementById('billedHours').value) || 0;
            const pharmacy = document.getElementById('pharmacyId').value;
            const secondaryCpt = document.getElementById('secondaryCpt').value;

            // Sensitivity multipliers
            const rMult = parseInt(document.getElementById('rulesSensitivity').value) / 50;
            const mMult = parseInt(document.getElementById('mlSensitivity').value) / 50;

            resetPipelineSteps();

            let totalScore = 0;
            let shapAttributions = [];

            // 1. Pre-Pay Check
            setTimeout(() => {
                updateStepUI('step-prepay', 'PASS', 'success');
            }, 150);

            // 2. Rules Engine
            setTimeout(() => {
                if (hours > 24) {
                    const pts = Math.round(45 * rMult);
                    totalScore += pts;
                    shapAttributions.push({ feature: 'Phantom Hours (>24h in 1 Calendar Day)', impact: `+${pts} pts`, level: 'rose' });
                } else if (hours > 12) {
                    const pts = Math.round(18 * rMult);
                    totalScore += pts;
                    shapAttributions.push({ feature: 'Excessive Daily Billed Hours (>12h)', impact: `+${pts} pts`, level: 'amber' });
                }

                if ((cpt === '95907' && secondaryCpt === '95904') || (cpt === '95904' && secondaryCpt === '95907')) {
                    const pts = Math.round(35 * rMult);
                    totalScore += pts;
                    shapAttributions.push({ feature: 'Unbundling Violation (Nerve Conduction Pair)', impact: `+${pts} pts`, level: 'rose' });
                }

                updateStepUI('step-rules', totalScore > 0 ? 'RULES TRIGGERED' : 'CLEAN', totalScore > 0 ? 'warn' : 'success');
            }, 300);

            // 3. ML Outlier Isolation
            setTimeout(() => {
                if (cpt === '99215' && specialty === 'Dermatology') {
                    const pts = Math.round(38 * mMult);
                    totalScore += pts;
                    shapAttributions.push({ feature: 'Upcoding Outlier: Level 5 E/M vs Specialty Peer Baseline', impact: `+${pts} pts`, level: 'rose' });
                } else if (billed > 1200 && !cpt.startsWith('95')) {
                    const pts = Math.round(22 * mMult);
                    totalScore += pts;
                    shapAttributions.push({ feature: 'Billed Fee Outlier (Z-Score > 3.1 vs Local Median)', impact: `+${pts} pts`, level: 'amber' });
                }

                updateStepUI('step-ml', totalScore > 20 ? 'ANOMALY DETECTED' : 'NORMAL', totalScore > 20 ? 'warn' : 'success');
            }, 450);

            // 4. Graph Discovery
            setTimeout(() => {
                let isCollusion = false;
                if (pharmacy.includes('RING') || pharmacy.includes('SUSPECT') || npi === '1002938192') {
                    isCollusion = true;
                    const pts = 35;
                    totalScore += pts;
                    shapAttributions.push({ feature: 'Collusion Network Ring (High-Density Kickbacks)', impact: `+${pts} pts`, level: 'purple' });
                }

                updateStepUI('step-graph', isCollusion ? 'RING DETECTED' : 'NO RING', isCollusion ? 'danger' : 'success');
                
                currentGraphMode = isCollusion ? 'kickback' : 'normal';
                renderGraph(currentGraphMode);

                renderGaugeAndSHAP(totalScore, shapAttributions, { npi, cpt, billed });
            }, 600);
        }

        // HTML5 Canvas Interactive Graph Engine (With Drag & Drop)
        function initGraphCanvas() {
            canvas = document.getElementById('graphCanvas');
            ctx = canvas.getContext('2d');

            function resizeCanvas() {
                canvas.width = canvas.offsetWidth;
                canvas.height = canvas.offsetHeight;
                renderGraph(currentGraphMode);
            }
            resizeCanvas();
            window.addEventListener('resize', resizeCanvas);

            // Drag Events
            canvas.addEventListener('mousedown', (e) => {
                const rect = canvas.getBoundingClientRect();
                const mx = e.clientX - rect.left;
                const my = e.clientY - rect.top;

                graphNodes.forEach(n => {
                    const dist = Math.hypot(n.x - mx, n.y - my);
                    if (dist < n.radius + 5) {
                        draggedNode = n;
                    }
                });
            });

            canvas.addEventListener('mousemove', (e) => {
                if (draggedNode) {
                    const rect = canvas.getBoundingClientRect();
                    draggedNode.x = e.clientX - rect.left;
                    draggedNode.y = e.clientY - rect.top;
                    drawGraphScene();
                }
            });

            canvas.addEventListener('mouseup', () => { draggedNode = null; });
        }

        function renderGraph(mode) {
            if (!canvas) return;
            const w = canvas.width;
            const h = canvas.height;

            graphNodes = [];
            graphEdges = [];

            if (mode === 'kickback') {
                document.getElementById('graphRiskTag').textContent = 'CRITICAL: Collusion Kickback Ring';
                document.getElementById('graphRiskTag').className = 'text-[10px] font-mono px-2 py-1 rounded bg-rose-500/10 text-pearl-rose border border-rose-500/30';

                const p1 = { id: 1, x: w * 0.25, y: h * 0.5, radius: 18, color: '#06b6d4', label: 'Dr. Mercer (NPI)' };
                const rx = { id: 2, x: w * 0.75, y: h * 0.5, radius: 18, color: '#a855f7', label: 'Rx Kickback Direct' };
                
                graphNodes.push(p1, rx);

                for (let i = 0; i < 4; i++) {
                    const pt = { id: 10 + i, x: w * 0.5, y: h * (0.2 + i * 0.2), radius: 10, color: '#10b981', label: `Pt #${101 + i}` };
                    graphNodes.push(pt);
                    graphEdges.push({ from: p1, to: pt, color: '#f43f5e', dashed: true });
                    graphEdges.push({ from: pt, to: rx, color: '#f43f5e', dashed: false });
                }
            } else {
                document.getElementById('graphRiskTag').textContent = 'Topology: Normal Network';
                document.getElementById('graphRiskTag').className = 'text-[10px] font-mono px-2 py-1 rounded bg-slate-800 text-slate-400 border border-slate-700';

                const p1 = { id: 1, x: w * 0.2, y: h * 0.5, radius: 16, color: '#06b6d4', label: 'Standard NPI' };
                const pt1 = { id: 2, x: w * 0.55, y: h * 0.3, radius: 10, color: '#10b981', label: 'Patient A' };
                const pt2 = { id: 3, x: w * 0.55, y: h * 0.7, radius: 10, color: '#10b981', label: 'Patient B' };
                const rx1 = { id: 4, x: w * 0.85, y: h * 0.3, radius: 14, color: '#a855f7', label: 'CVS Pharmacy' };
                const rx2 = { id: 5, x: w * 0.85, y: h * 0.7, radius: 14, color: '#a855f7', label: 'Walgreens Rx' };

                graphNodes.push(p1, pt1, pt2, rx1, rx2);
                graphEdges.push(
                    { from: p1, to: pt1, color: '#334155', dashed: false },
                    { from: p1, to: pt2, color: '#334155', dashed: false },
                    { from: pt1, to: rx1, color: '#334155', dashed: false },
                    { from: pt2, to: rx2, color: '#334155', dashed: false }
                );
            }

            drawGraphScene();
        }

        function drawGraphScene() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);

            // Draw Edges
            graphEdges.forEach(e => {
                ctx.beginPath();
                ctx.moveTo(e.from.x, e.from.y);
                ctx.lineTo(e.to.x, e.to.y);
                ctx.strokeStyle = e.color;
                ctx.lineWidth = 2;
                if (e.dashed) ctx.setLineDash([4, 4]);
                else ctx.setLineDash([]);
                ctx.stroke();
            });

            // Draw Nodes
            graphNodes.forEach(n => {
                ctx.beginPath();
                ctx.arc(n.x, n.y, n.radius, 0, Math.PI * 2);
                ctx.fillStyle = n.color;
                ctx.fill();
                ctx.strokeStyle = '#0f172a';
                ctx.lineWidth = 2.5;
                ctx.stroke();

                ctx.fillStyle = '#94a3b8';
                ctx.font = '10px Plus Jakarta Sans';
                ctx.textAlign = 'center';
                ctx.fillText(n.label, n.x, n.y + n.radius + 12);
            });
        }
